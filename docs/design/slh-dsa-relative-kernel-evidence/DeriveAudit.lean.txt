import RelativeKernel

/-! # Complete signature-outcome specialization of relative tape averaging -/

open OracleSpec OracleComp SignatureAlg

namespace SLHDSA.Security.CoverageDeferred

private theorem deriveOutcomeSpecialization :
  ∀ {vp : ValidatedParams} (core : CorePrimitives vp.params)
    [SampleableType core.Y] [SampleableType (Bytes vp.params.m)]
    [DecidableEq core.Y] [DecidableEq core.PkSeed] [DecidableEq core.AdrsKey]
    [DecidableEq core.SkPrf],
  ∀ (pkSeed : core.PkSeed)
    (policy : SigningView.Event (labSpec core) (List Byte) → Nat)
    (program : OracleComp (SigningView.Events.Spec (labSpec core) (List Byte))
      (DeriveOutcome core))
    (initialCost : Nat) (initialPhase : SigningView.Phase)
    (signerLength forgerLength : Nat),
    let initial : S core := (((∅, ∅), []), (initialCost, initialPhase))
    let lengths : Coverage.TapeClass → Nat := fun
      | .signer => signerLength
      | .forger => forgerLength
      | .other => 0
    letI : MeasurableSpace
      (((DeriveOutcome core) × S core) × (Coverage.jointSpec core).QueryCache) := ⊤
    𝒟[(do
      let tapes ← AnswerTape.tapeFamily (Coverage.tapeClassRange core) lengths
      let result ← (simulateQ (Coverage.roleRunImpl core)
        ((simulateQ (relative core pkSeed policy) program).run initial)).run
          ((∅, tapes), (AnswerTape.ClassPos.init, ∅))
      return (result.1, result.2.1.1))] =
      𝒟[(fun result => (result, nativeCache core result.2.1)) <$>
        (simulateQ
          ((SigningView.Events.handlerWithEvents (Request := List Byte)
            (slhGraph core pkSeed).deferredImpl).extendState
              (phaseCostAux core policy)) program).run initial] := by
  intro vp core _ _ _ _ _ _ pkSeed policy program initialCost initialPhase
    signerLength forgerLength
  exact evalDist_relative_average_eq_deferred core pkSeed policy program initialCost
    initialPhase signerLength forgerLength

#print axioms deriveOutcomeSpecialization

end SLHDSA.Security.CoverageDeferred
