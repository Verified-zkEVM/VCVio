private theorem audit_prEvent_cnt_overwrite
    {vp : ValidatedParams} (core : CorePrimitives vp.params)
    [SampleableType core.Y] [SampleableType (Bytes vp.params.m)] [DecidableEq core.Y]
    [DecidableEq core.PkSeed] [DecidableEq core.AdrsKey] [DecidableEq core.SkPrf]
    (tracked : TapeClass) {α : Type} (program : OracleComp (roleSpec core) α)
    (s : RoleState core) (index cell : Nat) (value : tapeClassRange core tracked)
    (horder : index ≤ cell) :
    Pr{let z ← ((simulateQ (roleRunImpl core) program).run
      ((s.1.1, Function.update s.1.2 tracked ((s.1.2 tracked).set cell value)), s.2))}[
        s.2.1.cnt tracked + index < z.2.2.1.cnt tracked] =
    Pr{let z ← (simulateQ (roleRunImpl core) program).run s}[
      s.2.1.cnt tracked + index < z.2.2.1.cnt tracked] := by
  exact prEvent_cnt_overwrite core tracked program s index cell value horder

end SLHDSA.Security.Coverage.NativeCausality
namespace SLHDSA.Security.Coverage

private theorem audit_prEvent_roleRun_forgerCnt_set_ge
    {vp : ValidatedParams} (core : CorePrimitives vp.params)
    [SampleableType core.Y] [SampleableType (Bytes vp.params.m)] [DecidableEq core.Y]
    [SampleableType core.SkSeed] [SampleableType core.SkPrf]
    [DecidableEq core.PkSeed] [DecidableEq core.AdrsKey] [DecidableEq core.SkPrf]
    (e : core.SkSeed ≃ core.Y)
    (kappa : PublicKeyCore core → core.Y) (pkSeed : core.PkSeed)
    (adv : UnforgeableAdversary
      (romScheme core e (fun pk => pure (kappa pk)) (pure pkSeed)))
    (L : (j : TapeClass) → List (tapeClassRange core j)) (hL : L .other = [])
    (index cell : Nat) (value : Bytes vp.params.m) (horder : index ≤ cell) :
    Pr{let z ← (roleRun core adv pkSeed
      (Function.update L TapeClass.forger ((L TapeClass.forger).set cell value)))}[
        index < z.2.2.1.cnt TapeClass.forger] =
      Pr{let z ← roleRun core adv pkSeed L}[
        index < z.2.2.1.cnt TapeClass.forger] := by
  exact prEvent_roleRun_forgerCnt_set_ge core e kappa pkSeed adv L hL index cell value horder

/-- The signer second-attempt observation is invariant under later signer-tape overwrites. -/
private theorem audit_signerCnt_two_set_ge
    {vp : ValidatedParams} (core : CorePrimitives vp.params)
    [SampleableType core.Y] [SampleableType (Bytes vp.params.m)] [DecidableEq core.Y]
    [DecidableEq core.PkSeed] [DecidableEq core.AdrsKey] [DecidableEq core.SkPrf]
    {α : Type} (program : OracleComp (roleSpec core) α)
    (L : (j : TapeClass) → List (tapeClassRange core j))
    (cell : Nat) (value : Bytes vp.params.m) (horder : 1 ≤ cell) :
    Pr{let z ← ((simulateQ (roleRunImpl core) program).run
      ((∅, Function.update L TapeClass.signer ((L TapeClass.signer).set cell value)),
        (AnswerTape.ClassPos.init, ∅)))}[2 ≤ z.2.2.1.cnt TapeClass.signer] =
    Pr{let z ← ((simulateQ (roleRunImpl core) program).run
      ((∅, L), (AnswerTape.ClassPos.init, ∅)))}[2 ≤ z.2.2.1.cnt TapeClass.signer] := by
  simpa only [AnswerTape.ClassPos.init, Nat.zero_add, Nat.succ_le_iff] using
    NativeCausality.prEvent_cnt_overwrite core TapeClass.signer program
      (((∅, L), (AnswerTape.ClassPos.init, ∅)) : RoleState core) 1 cell value horder

end SLHDSA.Security.Coverage
