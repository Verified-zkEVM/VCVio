import StoppedExpectedCharge

/-! # Closed recovery checks for coinciding stopping and hazard predicates -/

open OracleComp OracleSpec MeasureTheory
open scoped ENNReal

namespace OracleComp

private theorem recover_original_potential_step
    {ι σ : Type} {spec : OracleSpec ι}
    (handler : QueryImpl spec (StateT σ ProbComp))
    (bad : σ → Prop) [DecidablePred bad] (potential : σ → ℝ≥0∞)
    (charged : IsPotentialStep handler bad potential)
    (fallback : (t : spec.Domain) → spec.Range t) :
    IsPotentialStep (handler.stoppedAtBad bad fallback) bad potential :=
 by
  exact IsPotentialStep.stoppedAtBad handler bad bad potential charged fallback

private theorem recover_original_expected_endpoint
    {ι σ α : Type} {spec : OracleSpec ι}
    (handler : QueryImpl spec (StateT σ ProbComp))
    (bad : σ → Prop) [DecidablePred bad] (potential : σ → ℝ≥0∞)
    (charged : IsPotentialStep handler bad potential)
    (fallback : (t : spec.Domain) → spec.Range t)
    (program : OracleComp spec α) (state : σ) (initiallyGood : ¬bad state) :
    Pr{let z ← (simulateQ (handler.stoppedAtBad bad fallback) program).run state}[bad z.2] +
      potential state ≤
        ∫⁻ r, r ∂𝒟[(fun z => potential z.2) <$>
          (simulateQ (handler.stoppedAtBad bad fallback) program).run state] :=
 by
  exact prEvent_stopped_run_add_initial_potential_le_expected_final
    handler bad bad potential charged fallback program state initiallyGood

end OracleComp

#print axioms OracleComp.recover_original_potential_step
#print axioms OracleComp.recover_original_expected_endpoint
